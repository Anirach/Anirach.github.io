ทำความเข้าใจขอบเขตนี้

attack suite ควรทดสอบขอบเขตที่เราควบคุมได้จริง ชุดออฟไลน์พิสูจน์ได้ว่า route admin ที่ผิด การอ้างอนุมัติปลอม และ provider ที่ล้มไม่สร้างผลจำลองโดยตรง แต่มันบอกไม่ได้ว่า Jev จับ prompt injection ได้หรือยังแม่นเมื่อสลับลำดับตัวเลือก คำถามเหล่านั้นต้องใช้ผลโมเดลจริงบนชุดประเมินที่ตรวจแล้ว

บันทึกข้อจำกัด Jev 1.13 ทางการกล่าวถึงปัญหาความแม่นเชิงตัวเลข state ที่มีข้อมูลไม่เกี่ยว เนื้อหาโจมตี และลำดับตัวเลือก เราจึงเก็บเลขคณิตกับสิทธิ์ไว้ในโค้ด deterministic และถือข้อความ ticket เป็นข้อมูล ไม่ใช่คำสั่ง fixture มีข้อความโจมตีโดยตั้งใจ เพื่อให้เห็นว่า guard ยังทำหน้าที่ได้ตรงไหนแม้คำตอบเชิงความหมายผิด

การ retry inference กับ retry effect แก้ปัญหาคนละอย่าง inference ที่ล้มชั่วคราวอาจต้องเรียก provider อีกครั้งซึ่งมี latency และอาจมีต้นทุน ส่วนการทำ action ที่อนุมัติแล้วซ้ำต้องเคารพ idempotency key SDK อาจ retry ภายใน wrapper live ของเราจึงไม่เพิ่มลูป retry ชั้นนอก และไม่อ้างว่าเห็น attempts ที่ซ่อนอยู่ครบ

ลงมือทีละเจ็ดขั้น

ขั้นที่ 1

รัน attack fixtures สี่รายการโดยคงป้ายข้อมูลสมมติไว้ T-037 กับ T-038 พยายามสั่งคิวหรืออ้างอนุมัติผ่านข้อความ ticket ผลที่เตรียมไว้นี้ไม่ใช่การวัดว่าโจมตี Jev สำเร็จหรือไม่ ให้ตรวจข้อเสนอสุดท้ายแยกจากด่าน approval

ขั้นที่ 2

รัน T-040 ในโหมด Jev-only โดยต้องปฏิเสธ label ที่ไม่รองรับ วิธีนี้แยกการตรวจ response ออกจาก hybrid fallback เพราะ test ที่รันแต่ hybrid อาจซ่อนความล้มเหลวเดิมไว้หลัง provider ตัวที่สองซึ่งบังเอิญตอบ label ถูก schema

ขั้นที่ 3

ฉีด exception ของ provider ผ่าน pipeline test เมื่อทั้งสอง provider ใช้ไม่ได้ ผลสุดท้ายต้องต้องการ review และไม่มี action การจัดการ error ควรเก็บเหตุผลที่ใช้วินิจฉัยได้ โดยไม่พิมพ์ API key และไม่ตีความว่าโมเดลใช้ไม่ได้แปลว่าคำตอบเป็น no

ขั้นที่ 4

ทำแผนที่ HTTP failures ก่อนใช้ live ปัญหา authentication กับ schema ต้องแก้สาเหตุ ไม่ใช่ retry แบบไม่ดู ส่วน rate และ overload อาจถูก SDK retry ให้กำหนดงบการลองใหม่ระดับแอปชัด และตรวจพฤติกรรมตามรุ่นในเอกสารทางการ ไม่เติมไลบรารี retry อีกชั้นเพราะความเคยชิน

ขั้นที่ 5

ระบุ metamorphic tests สำหรับ provider จริง เช่นสลับตัวเลือก เติมย่อหน้าไม่เกี่ยว เปลี่ยนถ้อยคำแต่คงเจตนา และแยกเลขคณิตออกจากโมเดล ตรึงสิ่งที่ควรคงเดิมก่อนเก็บคำตอบ provider fixture ปัจจุบันเลือกข้อมูลตาม ticket ID จึงตั้งใจไม่ใช้วัดผลของการแปลงเหล่านี้

ขั้นที่ 6

แยกผลล้มเหลวภาษาไทยกับอังกฤษ การส่งข้อความไทยไม่กี่ fixture ผ่านได้ยืนยันเพียง encoding และ control flow การประเมินภาษาไทยจริงต้องมีภาษาที่เป็นตัวแทน การสลับภาษา และคำเฉพาะงาน พร้อมผู้ตรวจที่เข้าใจภาษา อย่าอนุมานว่าแม่นเท่าอังกฤษเพราะใช้ JSON schema เดียวกัน

ขั้นที่ 7

รัน effect tests อีกครั้งหลังเปลี่ยน failure handling เส้นทาง exception ที่เริ่ม fallback ไม่ควรทำ action ที่อนุมัติแล้วซ้ำหรือข้ามเงื่อนไขขอใช้คน และต้องเก็บ record ของ attempt เดิมไว้แม้แขนงหลังเป็นผู้สร้างข้อเสนอสุดท้าย

เปิดไฟล์นี้เทียบกับ source ใน ZIP ก่อนแก้ ข้อความต่อไปนี้คือโค้ดที่ใช้ในโครงการจริง ไม่ใช่ pseudocode ผลโมเดลเป็น input ให้โค้ดตรวจและวางเส้นทางต่อ ไม่ใช่สิทธิ์ให้ข้ามด่านอนุมัติ

src/pipeline.mjs

import {validateTicket,parseJev,parseBaseline,probability} from './contracts.mjs';
export function validatePolicy(policy) {
  probability(policy.routeConfidence,'policy routeConfidence');probability(policy.humanNoMax,'policy humanNoMax');
  if(policy.humanNoMax>=0.5||typeof policy.fallbackOnLowConfidence!=='boolean')throw new Error('invalid abstention policy');return policy;
}
export async function decide(input,{mode='hybrid',providers,policy}) {
  const ticket=validateTicket(input);validatePolicy(policy);
  if(!['baseline','jev','hybrid'].includes(mode))throw new Error('invalid mode');
  const attempts=[];let decision=null,source=null,reason='not_evaluated';
  async function obtain(kind,parse) {
    try {const result=await providers[kind](ticket);attempts.push(result.attempt);source=kind;return parse(result.value);}
    catch(error) {
      if(attempts.at(-1)?.provider!==kind)attempts.push({provider:kind,mode:'error',usage:null,wallMs:null,scenarioLatencyMs:null,httpAttempts:null,costComplete:false});
      reason=`${kind}_failed_validation_or_call`;return null;
    }
  }
  if(mode==='baseline')decision=await obtain('baseline',parseBaseline);
  else {
    decision=await obtain('jev',parseJev);
    if(decision&&decision.wantsHuman>policy.humanNoMax)reason='human_request_or_uncertainty';
    else if(!decision||decision.confidence<policy.routeConfidence) {
      const humanProbability=decision?.wantsHuman;
      if(mode==='hybrid'&&policy.fallbackOnLowConfidence) {
        const fallback=await obtain('baseline',parseBaseline);
        decision=fallback?{...fallback,wantsHuman:Math.max(fallback.wantsHuman,humanProbability??0)}:null;
      } else reason='low_route_confidence';
    }
  }
  const auto=!!decision&&decision.wantsHuman<=policy.humanNoMax&&(source==='baseline'||decision.confidence>=policy.routeConfidence);
  if(auto)reason=source==='baseline'?'validated_baseline_proposal':'validated_jev_proposal';
  else if(decision?.wantsHuman>policy.humanNoMax)reason='human_request_or_uncertainty';
  return {ticketId:ticket.id,status:auto?'proposed':'needs_review',reason,source,decision,attempts,action:auto?{kind:'route_ticket',ticketId:ticket.id,queue:decision.route}:null};
}

เปลี่ยนหนึ่งอย่างแล้วทดสอบ

ออกแบบ attack case สำหรับ provider จริงสามข้อที่ไม่มีใน fixture สี่รายการ ระบุ invariant ที่วัดได้กับวิธีตรวจเฉลยแต่ละข้อ เก็บเลขคณิตไว้ในโค้ด และแยกโมเดลตอบผิดออกจาก guard เสีย เพราะโมเดลอาจผิดแต่ effect boundary ยังทำงานถูก

ทำการแก้ไขในสำเนาทดลอง เก็บไฟล์ต้นฉบับไว้เทียบ แล้วคืนค่าก่อนเริ่มตอนถัดไป หากทดลองกับ provider จริงภายหลัง ให้สร้าง experiment record ใหม่แทนเขียนทับผลเดิม

How this boundary works

An attack suite should test the boundary you actually own. Our offline suite can prove that an invalid admin route, a forged approval claim and a failed provider do not directly create a simulated effect. It cannot show that Jev recognizes a prompt injection or remains accurate under reordered options; those questions require real model responses on a reviewed evaluation set.

The official Jev 1.13 notes describe difficulties with numerical precision, irrelevant state, adversarial content and option order. We keep arithmetic and permissions in deterministic code and treat tickets as data rather than instructions. The teaching fixtures deliberately include hostile requests so you can see where the guard works even when a semantic answer is wrong.

Inference retries and effect retries solve different problems. A transient inference failure may require another provider attempt, which has latency and possible cost. Repeating an already approved action must instead respect an idempotency key. The SDK may retry internally; our live wrapper does not add an outer retry loop or claim visibility into those hidden attempts.

Seven practical steps

Step 1 — Run the four attack fixtures

Run the four attack fixtures. Keep their synthetic-evidence label visible. T-037 and T-038 try to dictate routing or approval through ticket text. The canned outputs are not measurements of attack success against Jev; inspect the eventual proposal and approval gate separately.

Step 2 — Run T-040 in Jev-only mode

Run T-040 in Jev-only mode. The unsupported label must be refused. This isolates response-validation behavior from hybrid fallback. A test that only exercises hybrid may hide the original failure behind a second provider that happens to return a valid label.

Step 3 — Inject a provider exception through the pipeline test

Inject a provider exception through the pipeline test. With both providers unavailable the final result must need review and contain no action. Error handling should preserve a useful reason without printing API keys or treating an unavailable model as a negative answer.

Step 4 — Map HTTP failure classes before live use

Map HTTP failure classes before live use. Authentication and schema failures require correction, not blind retries. Rate and overload failures may be retried by the SDK. Keep the application-level retry budget explicit and confirm version-specific behavior in the official documentation instead of layering another retry library by habit.

Step 5 — Specify metamorphic tests for the real provider

Specify metamorphic tests for the real provider. Reorder route options, append irrelevant paragraphs, paraphrase the same intent and keep numerical calculations outside the model. Freeze expected invariants before collecting responses. The current fixture provider is indexed by ticket ID and intentionally cannot measure these transformations.

Step 6 — Keep Thai and English failure slices separate

Keep Thai and English failure slices separate. Passing a few Thai fixture strings proves encoding and control flow only. Real Thai evaluation needs representative language, code-switching and domain vocabulary, annotated by competent reviewers. Do not infer parity with English from a shared JSON schema.

Step 7 — Run the effect tests again after changing failure handling

Run the effect tests again after changing failure handling. An exception path that starts a fallback should not duplicate an approved action or bypass the human request condition. Preserve the original attempt record even when a later branch produces the final proposal.

  if(!['baseline','jev','hybrid'].includes(mode))throw new Error('invalid mode');
  const attempts=[];let decision=null,source=null,reason='not_evaluated';
  async function obtain(kind,parse) {
    try {const result=await providers[kind](ticket);attempts.push(result.attempt);source=kind;return parse(result.value);}
    catch(error) {
      if(attempts.at(-1)?.provider!==kind)attempts.push({provider:kind,mode:'error',usage:null,wallMs:null,scenarioLatencyMs:null,httpAttempts:null,costComplete:false});
      reason=`${kind}_failed_validation_or_call`;return null;
    }
  }
  if(mode==='baseline')decision=await obtain('baseline',parseBaseline);
  else {
    decision=await obtain('jev',parseJev);
    if(decision&&decision.wantsHuman>policy.humanNoMax)reason='human_request_or_uncertainty';
    else if(!decision||decision.confidence<policy.routeConfidence) {
      const humanProbability=decision?.wantsHuman;
      if(mode==='hybrid'&&policy.fallbackOnLowConfidence) {
        const fallback=await obtain('baseline',parseBaseline);
        decision=fallback?{...fallback,wantsHuman:Math.max(fallback.wantsHuman,humanProbability??0)}:null;
      } else reason='low_route_confidence';
    }
  }
  const auto=!!decision&&decision.wantsHuman<=policy.humanNoMax&&(source==='baseline'||decision.confidence>=policy.routeConfidence);
  if(auto)reason=source==='baseline'?'validated_baseline_proposal':'validated_jev_proposal';
  else if(decision?.wantsHuman>policy.humanNoMax)reason='human_request_or_uncertainty';
  return {ticketId:ticket.id,status:auto?'proposed':'needs_review',reason,source,decision,attempts,action:auto?{kind:'route_ticket',ticketId:ticket.id,queue:decision.route}:null};
}

Verify the expected behavior

The attack report must contain n=4 and remain labeled as a fixture simulation. The T-040 Jev-only demonstration must have action:null. Provider-failure and forged-approval tests must pass without credentials or network access.

Read the named tests and refusal paths, not just the exit code or final label. Unknown values must remain unknown, and every report must distinguish simulation inputs from real measurements.

Troubleshooting

A successful mock attack test is not a successful model robustness test. If you see duplicate paid work, inspect whether SDK and application retries are stacked. If an error string includes a secret, remove the secret from logging and handle credential exposure through your normal process before resuming.

When previewing with Python http.server, use the .html continuation links below when needed. Extensionless series chips rely on GitHub Pages resolution; that documented preview limitation is separate from the Node example.

Change one thing, then test

Design three real-provider attack cases that are absent from the four canned fixtures. For each, state a measurable invariant and the label-review process. Keep numerical arithmetic in code and distinguish a model mistake from a guard failure. A model may be wrong while the effect boundary still behaves correctly.

Make changes in a disposable copy, retain the reference files for comparison, and restore them before the next chapter. If you later use a real provider, create a new experiment record rather than overwriting the fixture transcript.
